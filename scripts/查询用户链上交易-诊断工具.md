### 方法2: 通过 RPC 查询 (需要代码)

如果您有 RPC 节点访问权限，可以使用以下 Node.js 脚本：

\`\`\`javascript
const { ethers } = require('ethers');

const provider = new ethers.providers.JsonRpcProvider('您的RPC节点');
const contractAddress = '您的空投合约地址';

// ABI 片段
const abi = [
  "event Claimed(address indexed user, uint256 amount)"
];

const contract = new ethers.Contract(contractAddress, abi, provider);

async function queryUserClaims(userAddress) {
  console.log(\`查询用户 \${userAddress} 的领取记录...\`);
  
  try {
    const filter = contract.filters.Claimed(userAddress);
    const events = await contract.queryFilter(filter, -100000); // 查询最近10万个区块
    
    console.log(\`找到 \${events.length} 个领取记录:\`);
    
    for (const event of events) {
      console.log(\`- 交易哈希: \${event.transactionHash}\`);
      console.log(\`  区块号: \${event.blockNumber}\`);
      console.log(\`  金额: \${ethers.utils.formatEther(event.args.amount)} RAT\`);
    }
    
    return events;
  } catch (error) {
    console.error('查询失败:', error.message);
    return [];
  }
}

// 查询两个用户
(async () => {
  await queryUserClaims('0xf0dfddd1d74138280916d86702f3c1c66171045b');
  await queryUserClaims('0xe8c903de963a446c661071251762d328420ccd19');
})();
\`\`\`
